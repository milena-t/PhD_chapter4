library(tximport)
library(DESeq2)
library(readr)
library(ggplot2)

results_dir <- "/Users/bsarc/Desktop/salmon_results"

files <- list.files(results_dir, pattern="\\.sf$", full.names = TRUE)

names(files) <- tools::file_path_sans_ext(basename(files))

coldata<- read.csv("/Users/bsarc/Desktop/sample_metadata.csv", row.names = 1)

txi <- tximport(files, type = "salmon", txOut = TRUE)

dds <- DESeqDataSetFromTximport(txi, colData = coldata, design = ~ sex + age + tor_copies)

dds <- DESeq(dds)
res <- results(dds)

head(res[order(res$padj), ])


colData(dds)$age <- factor(colData(dds)$age)

colData(dds)$tor_copies <- factor(colData(dds)$tor_copies)

vsd <- vst(dds, blind = TRUE)  # blind=TRUE ignores design for unbiased PCA
# 6. PCA plot
tor<- plotPCA(vsd, intgroup = "tor_copies")
tor + ggtitle("PCA of samples by TOR copy number")

age <- plotPCA(vsd, intgroup = "age")
age + ggtitle("PCA of samples by age (days)")+ scale_color_brewer(palette = "Set2")

sex <- plotPCA(vsd, intgroup = "sex")
sex + ggtitle("PCA of samples by sex")

sex <- plotPCA(vsd, intgroup = c("tor_copies","sex"))
sex + ggtitle("PCA of samples by number of TOR copies and sex") + scale_color_manual(values = c("1:F" = "firebrick", "1:M" = "darkblue","3:F" = "salmon", "3:M" = "lightblue"))



plotPCA(vsd, intgroup = c("tor_copies", "sex"))
age <- plotPCA(vsd, intgroup = "age")
plotPCA(vsd, intgroup = c("tor_copies", "age"))
tor + ggtitle("PCA of samples by TOR copy number")
